ClaraMed Radiology AI on Streamlit + JeevaCare link: Implementation Plan

Make the Streamlit app work again, add a "Radiology AI" page, and give JeevaCare's sidebar a link to it. No data is exchanged between the sites.

For agentic workers: REQUIRED SUB-SKILL: Use subagent-driven-development (recommended) or executing-plans to implement this plan task-by-task. Each task's implementation goes through test-driven-development, which is mandatory in this repo. Steps use checkbox (- [ ]) syntax for tracking.

Goal: A working ClaraMed Streamlit app with a Radiology AI page, and a ready-to-apply JeevaCare patch that links to it.

Architecture: A new streamlit_app/ folder holds the restored legacy app, a pure radiology_flow module (validate an upload, run the existing X-ray service, rank the findings) and a thin Streamlit page. Dependencies are a separate lean list. JeevaCare gets a sidebar link driven by VITE_CLARAMED_BASE_URL.

Tech Stack: Python 3.11, Streamlit (with its AppTest harness), PyTorch CPU, TorchXRayVision, pytest; React 18, vitest 0.34, react-dom/server for the JeevaCare test. Spec: .slash-cli/specs/2026-11-14-streamlit-radiology-link-design.md.

Global Constraints

  • ClaraMed and JeevaCare stay separate: no signed links, callbacks, shared accounts or stored results between them.
  • The upload exists only in memory and one temp file, deleted on every path; file names and pixel data are never logged.
  • Upload limit 10 MB (MAX_UPLOAD_BYTES = 10 * 1024 * 1024); only PNG and JPEG, detected by magic bytes.
  • The Streamlit app must not import or install celery, sqlalchemy, psycopg2, redis, alembic or boto3.
  • No secrets in git: only placeholders in the secrets template; the real values go into Streamlit Cloud's Secrets box.
  • Run Python tests with python -m pytest from the repo root (the venv is Python 3.11). JeevaCare tests use npx vitest run in client/.
  • Commits are held until the user asks; no step below commits or pushes.

def detect_file(content: bytes, kind: str) -> tuple[str, str]:
    """Return (detected type, extension) if `content` is a valid file of `kind`, else raise FetchError('bad_content')."""
    for magic, name, ext in _MAGIC:
        if content.startswith(magic):
            if name in _ALLOWED[kind]:
                return name, ext
            break
    raise FetchError("bad_content", f"file content is not a valid {kind} file")

# streamlit_app/radiology_flow.py
"""Validate an uploaded chest X-ray and run the radiology model on it. No Streamlit imports."""
import logging
import tempfile
from dataclasses import dataclass
from pathlib import Path

from backend.core.exceptions import ValidationError
from backend.services.url_fetcher import FetchError, detect_file

logger = logging.getLogger(__name__)

MAX_UPLOAD_BYTES = 10 * 1024 * 1024


class UploadError(Exception):
    """A problem the user can act on. `message` is safe to show: it never contains file names or contents."""

    def __init__(self, code: str, message: str):
        super().__init__(message)
        self.code = code
        self.message = message


@dataclass
class Finding:
    label: str
    score: float


@dataclass
class RadiologyOutcome:
    findings: list[Finding]          # every finding, highest score first
    disclaimer: str
    model_name: str
    seconds: float

    def top(self, n: int = 3) -> list[Finding]:
        return self.findings[:n]


def analyze_upload(data: bytes, service) -> RadiologyOutcome:
    """Validate `data` as a PNG/JPEG and analyse it. The bytes touch disk only in one temp file, always removed."""
    if not data:
        raise UploadError("empty", "The file is empty.")
    if len(data) > MAX_UPLOAD_BYTES:
        raise UploadError("too_large", "The file is larger than 10 MB.")
    try:
        _, extension = detect_file(data, "radiology")           # decided by content, never by file name
    except FetchError:
        raise UploadError("not_an_image", "This does not look like a PNG or JPEG image.") from None

    path = None
    try:
        with tempfile.NamedTemporaryFile(delete=False, suffix=extension) as tmp:
            tmp.write(data)
            path = Path(tmp.name)
        result = service.analyze_chest_xray(path)
    except ValidationError:
        raise UploadError("invalid_image", "This image could not be read. Try another PNG or JPEG.") from None
    except Exception:
        logger.exception("radiology analysis failed")           # server log only; no file name or content
        raise UploadError("analysis_failed", "The analysis failed. Please try again.") from None
    finally:
        if path is not None:
            path.unlink(missing_ok=True)

    findings = sorted(
        (Finding(f["label"], float(f["model_score"])) for f in result["findings"]),
        key=lambda f: f.score, reverse=True,
    )
    return RadiologyOutcome(findings, result["disclaimer"], result["model_name"],
                            float(result["processing_duration_seconds"]))

import sys
from pathlib import Path
from urllib.parse import urlparse

import streamlit as st  # noqa: E402

from streamlit_app.radiology_flow import RadiologyOutcome, UploadError, analyze_upload  # noqa: E402

st.set_page_config(page_title="Radiology AI | ClaraMed", page_icon="🩻", layout="centered")

RESULT_KEY = "radiology_result"


@st.cache_resource(show_spinner="Loading the X-ray model (the first visit after a restart takes a minute)...")
def get_service():
    from backend.services.radiology_service import RadiologyService   # heavy import: only when first needed

    service = RadiologyService()
    service.load_model()
    return service


def jeevacare_url() -> str | None:
    """The optional 'Back to JeevaCare' target; only an https URL is ever used."""
    try:
        url = str(st.secrets.get("JEEVACARE_URL", "")).strip()
    except Exception:
        return None
    parsed = urlparse(url)
    return url if parsed.scheme == "https" and parsed.netloc else None


def render_result(outcome: RadiologyOutcome) -> None:
    st.subheader("Results")
    for column, finding in zip(st.columns(3), outcome.top(3)):
        column.metric(finding.label.replace("_", " "), f"{finding.score:.0%}")
    st.dataframe(
        [{"Finding": f.label.replace("_", " "), "Model score": f.score} for f in outcome.findings],
        column_config={"Model score": st.column_config.ProgressColumn(min_value=0.0, max_value=1.0, format="%.2f")},
        hide_index=True,
    )
    st.caption(outcome.disclaimer)
    st.caption(f"Model: {outcome.model_name} | {outcome.seconds:.1f}s on CPU")


st.title("🩻 Radiology AI")
st.warning("This is AI model output, not a diagnosis. A qualified radiologist must interpret the image.")
st.info("ClaraMed is a separate service from JeevaCare. Your image is analysed in memory and is not saved "
        "or sent anywhere, including to JeevaCare.")

back = jeevacare_url()
if back:
    st.link_button("← Back to JeevaCare", back)

upload = st.file_uploader("Chest X-ray (PNG or JPEG, up to 10 MB)", type=["png", "jpg", "jpeg"])
if upload is None:
    st.session_state.pop(RESULT_KEY, None)
elif st.button("Analyze", type="primary"):
    try:
        with st.spinner("Analysing..."):
            st.session_state[RESULT_KEY] = analyze_upload(upload.getvalue(), get_service())
    except UploadError as error:
        st.session_state.pop(RESULT_KEY, None)
        st.error(error.message)

if RESULT_KEY in st.session_state:
    render_result(st.session_state[RESULT_KEY])

# tests/unit/test_streamlit_requirements.py
"""The Streamlit app installs ONLY streamlit_app/requirements.txt. Keep it complete and lean."""
import ast
import re
import sys
import tomllib
from pathlib import Path

ROOT = Path(__file__).resolve().parents[2]
APP = ROOT / "streamlit_app"
ENTRY_POINTS = [APP / "app.py", APP / "pages" / "1_Radiology_AI.py"]
FIRST_PARTY = {"backend", "streamlit_app", "chains", "database"}
IMPORT_TO_DIST = {"PIL": "pillow", "skimage": "scikit-image", "PyPDF2": "pypdf2", "yaml": "pyyaml"}
BACKEND_ONLY = {"celery", "sqlalchemy", "psycopg2", "psycopg2-binary", "redis", "alembic", "boto3"}


def _first_party_file(module: str) -> Path | None:
    top = module.split(".")[0]
    base = APP if top in {"chains", "database"} else ROOT
    path = base / Path(*module.split("."))
    for candidate in (path.with_suffix(".py"), path / "__init__.py"):
        if candidate.exists():
            return candidate
    return None


def third_party_imports() -> set[str]:
    seen, found, queue = set(), set(), list(ENTRY_POINTS)
    while queue:
        path = queue.pop()
        if path in seen:
            continue
        seen.add(path)
        for node in ast.walk(ast.parse(path.read_text())):
            modules = []
            if isinstance(node, ast.Import):
                modules = [a.name for a in node.names]
            elif isinstance(node, ast.ImportFrom) and node.level == 0 and node.module:
                modules = [node.module]
            for module in modules:
                top = module.split(".")[0]
                if top in FIRST_PARTY:
                    target = _first_party_file(module)
                    if target:
                        queue.append(target)
                elif top not in sys.stdlib_module_names:
                    found.add(top)
    return found


def requirement_names() -> set[str]:
    names = set()
    for line in (APP / "requirements.txt").read_text().splitlines():
        line = line.strip()
        if line and not line.startswith(("#", "-")):
            names.add(re.split(r"[<>=!~\[ ;]", line)[0].lower().replace("_", "-"))
    return names


def test_every_third_party_import_is_installed():
    needed = {IMPORT_TO_DIST.get(m, m).lower().replace("_", "-") for m in third_party_imports()}
    missing = sorted(needed - requirement_names())
    assert missing == [], f"add these to streamlit_app/requirements.txt: {missing}"


def test_backend_only_packages_are_not_installed():
    assert sorted(requirement_names() & BACKEND_ONLY) == []


def test_pytorch_comes_from_the_cpu_index():
    first = (APP / "requirements.txt").read_text().splitlines()[0]
    assert first.strip() == "--extra-index-url https://download.pytorch.org/whl/cpu"

Requirements: R10

  • EXISTS: backend/services/radiology_service.py RadiologyService

# scripts/measure_streamlit_memory.py
"""Peak memory of the Streamlit app's heaviest path: Streamlit + legacy libraries + model + one analysis.

Run:  python scripts/measure_streamlit_memory.py
Compare the last line with Streamlit Community Cloud's current memory limit (about 1 GB at the time of writing).
"""
import io
import resource
import sys
from pathlib import Path

sys.path.insert(0, str(Path(__file__).resolve().parent.parent))

BUDGET_MB = 800


def peak_mb() -> float:
    peak = resource.getrusage(resource.RUSAGE_SELF).ru_maxrss
    return peak / (1024 * 1024) if sys.platform == "darwin" else peak / 1024      # macOS bytes, Linux KB


def main() -> int:
    import langchain_groq, markdown, pandas, PyPDF2, streamlit   # noqa: F401,E401  (what app.py loads)
    print(f"after Streamlit + legacy libraries: {peak_mb():6.0f} MB")

    from PIL import Image
    from backend.services.radiology_service import RadiologyService
    from streamlit_app.radiology_flow import analyze_upload

    service = RadiologyService()
    service.load_model()
    print(f"after loading the X-ray model:      {peak_mb():6.0f} MB")

    buffer = io.BytesIO()
    Image.effect_noise((1024, 1024), 40).convert("L").save(buffer, "PNG")
    outcome = analyze_upload(buffer.getvalue(), service)
    print(f"after one analysis ({len(outcome.findings)} findings): {peak_mb():6.0f} MB")

    verdict = "WITHIN" if peak_mb() <= BUDGET_MB else "OVER"
    print(f"\n{verdict} the {BUDGET_MB} MB budget (peak {peak_mb():.0f} MB, single process, excludes the Streamlit server's own overhead)")
    return 0 if verdict == "WITHIN" else 1


if __name__ == "__main__":
    raise SystemExit(main())

// client/src/config/claramed.js: add inside the claraMedConfig object

  /**
   * URL of ClaraMed's "Radiology AI" page, or null when ClaraMed is not configured.
   * ClaraMed is a separate website; this is a plain link, so nothing about the user or a record is sent.
   * Read at call time (not cached) so tests and runtime config changes take effect.
   */
  getRadiologyAiUrl() {
    const base = (import.meta.env.VITE_CLARAMED_BASE_URL || '').trim().replace(/\/+$/, '');
    if (!base) return null;
    let parsed;
    try {
      parsed = new URL(base);
    } catch {
      return null;
    }
    const isLocalhost = parsed.hostname === 'localhost' || parsed.hostname === '127.0.0.1';
    if (parsed.protocol !== 'https:' && !(parsed.protocol === 'http:' && isLocalhost)) return null;
    return `${base}/Radiology_AI`;
  },
