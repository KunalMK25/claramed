JeevaCare Linking Backend Implementation Plan

Make ClaraMed callable by JeevaCare through the contract in JeevaCare's own integration guide: authenticated submit, status, and a signed, retried callback.

For agentic workers: REQUIRED SUB-SKILL: Use subagent-driven-development (recommended) or executing-plans to implement this plan task-by-task. Each task's implementation goes through test-driven-development, which is mandatory in this repo. Steps use checkbox (- [ ]) syntax for tracking.

Goal: ClaraMed accepts radiology and document analysis requests from a registered client (JeevaCare first), processes them with the existing Celery workers, and pushes HMAC-signed results to the client's webhook with retries.

Architecture: A thin integration layer (/api/v1/analyze, /api/v1/status/{jobId}) over the existing analysis models and Celery tasks. Per-client settings (webhook URL, secret, allowed file hosts) live on IntegrationClient. A finalizer deletes source files and schedules the callback; a dispatcher signs and delivers it.

Tech Stack: Python 3.11, FastAPI, SQLAlchemy 2 + Alembic, PostgreSQL 16, Celery + Redis, httpx, pytest. Spec: .slash-cli/specs/2026-11-14-jeevacare-linking-design.md.

Global Constraints

  • ClaraMed side only: no edits to the JeevaCare repo (siri05gowda-byte/Jeevacare).
  • File URLs: HTTPS only, host in the client's allowlist, resolved IP public, size ≤ MAX_UPLOAD_SIZE_MB (default 10), content type must match type.
  • Callback header: X-ClaraMed-Signature: sha256=<HMAC(secret, timestamp + "." + body)> plus X-ClaraMed-Timestamp.
  • Callback retries: up to 5 attempts, exponential backoff over about 30 minutes, then delivery_status="failed".
  • Source file deleted after analysis finishes; results kept RESULT_RETENTION_DAYS (default 30).
  • Patient name and date of birth are never accepted in URLs or query strings; logs carry opaque IDs only.
  • Python 3.11 (venv at ./venv); run tests with source venv/bin/activate && python -m pytest.
  • DB driver stays postgresql+psycopg2 (pinned in backend/core/config.py).

A8. Accept JeevaCare's documented field names (changes T6)

JeevaCare's flow sends {radiologyId, imageUrl, patientId, metadata} with no type. AnalyzeRequest therefore accepts imageUrl as an alias of fileUrl, patientId as an alias of patientRef, and infers type (radiology when radiologyId is present, document when documentId is present). Exactly one of the URL fields and exactly one of the ids is required.

class AnalyzeRequest(BaseModel):
    type: Optional[Literal["radiology", "document"]] = None
    radiologyId: Optional[str] = Field(default=None, min_length=1, max_length=255)
    documentId: Optional[str] = Field(default=None, min_length=1, max_length=255)
    fileUrl: Optional[HttpUrl] = None
    imageUrl: Optional[HttpUrl] = None
    patientRef: Optional[str] = Field(default=None, max_length=255)
    patientId: Optional[str] = Field(default=None, max_length=255)
    hospitalRef: Optional[str] = Field(default=None, max_length=255)
    metadata: dict[str, Any] = Field(default_factory=dict)

    @model_validator(mode="after")
    def _normalise(self):
        if (self.radiologyId is None) == (self.documentId is None):
            raise ValueError("send exactly one of radiologyId or documentId")
        inferred = "radiology" if self.radiologyId is not None else "document"
        if self.type not in (None, inferred):
            raise ValueError(f"type '{self.type}' does not match the id sent")
        self.type = inferred
        if (self.fileUrl is None) == (self.imageUrl is None):
            raise ValueError("send exactly one of fileUrl or imageUrl")
        return self

    @property
    def record_id(self) -> str:
        return self.radiologyId or self.documentId

    @property
    def file_url(self) -> str:
        return str(self.fileUrl or self.imageUrl)

    @property
    def patient_ref(self):
        return self.patientRef or self.patientId

The service uses req.file_url and req.patient_ref. # backend/core/signing.py
"""HMAC signing for outbound webhook callbacks."""
import hashlib
import hmac
import time

SIGNATURE_PREFIX = "sha256="
DEFAULT_TOLERANCE_SECONDS = 300


def sign_payload(secret: str, timestamp: int, body: bytes) -> str:
    """Return 'sha256=<hex>' over '<timestamp>.<body>'."""
    message = f"{int(timestamp)}.".encode() + body
    return SIGNATURE_PREFIX + hmac.new(secret.encode(), message, hashlib.sha256).hexdigest()


def verify_signature(
    secret: str,
    timestamp: int,
    body: bytes,
    signature: str,
    now: float | None = None,
    tolerance: int = DEFAULT_TOLERANCE_SECONDS,
) -> bool:
    current = time.time() if now is None else now
    if abs(current - int(timestamp)) > tolerance:
        return False
    expected = sign_payload(secret, int(timestamp), body)
    return hmac.compare_digest(expected, signature)

# backend/services/client_registry.py
"""Register integration clients (admin operation)."""
import re
import secrets
from dataclasses import dataclass
from urllib.parse import urlparse

from sqlalchemy.exc import IntegrityError
from sqlalchemy.orm import Session

from backend.core.security import generate_api_key, hash_api_key
from backend.models.integration_client import IntegrationClient

DEFAULT_SCOPES = ["analysis:create", "analysis:read"]
_HOST_RE = re.compile(r"^(\*\.)?([a-z0-9-]+\.)+[a-z0-9-]+$")


@dataclass
class RegisteredClient:
    client: IntegrationClient
    api_key: str
    webhook_secret: str


def _validate(webhook_url: str, hosts: list[str]) -> list[str]:
    parsed = urlparse(webhook_url)
    if parsed.scheme != "https" or not parsed.hostname:
        raise ValueError("webhook_url must be an https URL")
    if not hosts:
        raise ValueError("allowed_file_hosts must contain at least one hostname")
    cleaned = [h.strip().lower() for h in hosts]
    for host in cleaned:
        if not _HOST_RE.match(host):
            raise ValueError(f"allowed_file_hosts entry is not a bare hostname: {host!r}")
    return cleaned


def register_client(
    db: Session,
    *,
    name: str,
    webhook_url: str,
    allowed_file_hosts: list[str],
    scopes: list[str] | None = None,
    description: str | None = None,
) -> RegisteredClient:
    hosts = _validate(webhook_url, allowed_file_hosts)
    api_key = generate_api_key()
    webhook_secret = secrets.token_hex(32)
    client = IntegrationClient(
        client_name=name,
        description=description,
        hashed_api_key=hash_api_key(api_key),
        scopes=list(scopes or DEFAULT_SCOPES),
        webhook_url=webhook_url,
        webhook_secret=webhook_secret,
        allowed_file_hosts=hosts,
        active=True,
    )
    db.add(client)
    try:
        db.commit()
    except IntegrityError as exc:
        db.rollback()
        raise ValueError(f"client {name!r} already exists") from exc
    return RegisteredClient(client=client, api_key=api_key, webhook_secret=webhook_secret)

"""Download a client-supplied file URL safely (SSRF-hardened).

Known limit (D6): DNS is resolved and checked, then httpx connects by hostname, so a
DNS-rebinding window remains. The per-client host allowlist is the primary control.
"""
import ipaddress
import socket
from dataclasses import dataclass
from urllib.parse import urljoin, urlparse

import httpx


class FetchError(Exception):
    def __init__(self, code: str, message: str):
        super().__init__(message)
        self.code = code
        self.message = message


@dataclass
class FetchedFile:
    content: bytes
    kind_detected: str  # "png" | "jpeg" | "pdf"
    extension: str


_MAGIC = [
    (b"\x89PNG\r\n\x1a\n", "png", ".png"),
    (b"\xff\xd8\xff", "jpeg", ".jpg"),
    (b"%PDF-", "pdf", ".pdf"),
]
_ALLOWED = {"radiology": {"png", "jpeg"}, "document": {"pdf"}}
_TIMEOUT = httpx.Timeout(15.0, connect=5.0)


def host_allowed(host: str, allowed: list[str]) -> bool:
    host = host.lower()
    for pattern in allowed:
        pattern = pattern.lower()
        if pattern.startswith("*."):
            if host.endswith(pattern[1:]) and host != pattern[2:]:
                return True
        elif host == pattern:
            return True
    return False


def default_resolver(host: str) -> list[str]:
    try:
        infos = socket.getaddrinfo(host, 443, proto=socket.IPPROTO_TCP)
    except socket.gaierror as exc:
        raise FetchError("fetch_failed", f"cannot resolve host: {exc}") from exc
    return [info[4][0] for info in infos]


def _is_public(ip_text: str) -> bool:
    ip = ipaddress.ip_address(ip_text)
    if isinstance(ip, ipaddress.IPv6Address) and ip.ipv4_mapped:
        ip = ip.ipv4_mapped
    return ip.is_global


def _validate(url: str, allowed_hosts: list[str], resolver) -> str:
    try:
        parsed = urlparse(url)
        host, port = parsed.hostname, parsed.port
    except ValueError as exc:
        raise FetchError("invalid_url", "URL is not valid") from exc
    if parsed.scheme != "https":
        raise FetchError("invalid_scheme", "only https URLs are accepted")
    if not host or parsed.username or parsed.password or port not in (None, 443):
        raise FetchError("invalid_url", "URL must be a plain https URL on the default port")
    if not host_allowed(host, allowed_hosts):
        raise FetchError("host_not_allowed", "host is not in the client's allowlist")
    addresses = resolver(host)
    if not addresses or not all(_is_public(a) for a in addresses):
        raise FetchError("blocked_address", "host resolves to a non-public address")
    return url


def _detect(content: bytes, kind: str) -> tuple[str, str]:
    for magic, name, ext in _MAGIC:
        if content.startswith(magic):
            if name in _ALLOWED[kind]:
                return name, ext
            break
    raise FetchError("bad_content", f"file content is not a valid {kind} file")


def fetch_file(
    url: str,
    *,
    allowed_hosts: list[str],
    kind: str,
    max_bytes: int,
    client: httpx.Client | None = None,
    resolver=None,
    max_redirects: int = 3,
) -> FetchedFile:
    if kind not in _ALLOWED:
        raise ValueError(f"unknown kind: {kind}")
    resolver = resolver or default_resolver
    owns_client = client is None
    client = client or httpx.Client(timeout=_TIMEOUT, follow_redirects=False)
    try:
        current = url
        for _hop in range(max_redirects + 1):
            _validate(current, allowed_hosts, resolver)
            try:
                with client.stream("GET", current) as response:
                    if response.is_redirect:
                        location = response.headers.get("location")
                        if not location:
                            raise FetchError("fetch_failed", "redirect without location")
                        current = urljoin(current, location)
                        continue
                    if response.status_code != 200:
                        raise FetchError("fetch_failed", f"upstream returned {response.status_code}")
                    declared = response.headers.get("content-length")
                    if declared and declared.isdigit() and int(declared) > max_bytes:
                        raise FetchError("too_large", "file exceeds the size limit")
                    buffer = bytearray()
                    for chunk in response.iter_bytes():
                        buffer.extend(chunk)
                        if len(buffer) > max_bytes:
                            raise FetchError("too_large", "file exceeds the size limit")
            except httpx.HTTPError as exc:
                raise FetchError("fetch_failed", f"download failed: {type(exc).__name__}") from exc
            name, ext = _detect(bytes(buffer), kind)
            return FetchedFile(content=bytes(buffer), kind_detected=name, extension=ext)
        raise FetchError("redirect_limit", "too many redirects")
    finally:
        if owns_client:
            client.close()

    """Return (analysis, created). Raises FetchError on intake failure (nothing is persisted)."""
    existing = _find_existing(db, client, req.type, req.record_id)
    if existing:
        return existing, False

    fetched = fetch_file(
        str(req.fileUrl),
        allowed_hosts=list(client.allowed_file_hosts or []),
        kind=req.type,
        max_bytes=settings.MAX_UPLOAD_SIZE_MB * 1024 * 1024,
    )

    analysis = MedicalAnalysis(
        analysis_type=AnalysisType.XRAY if req.type == "radiology" else AnalysisType.DOCUMENT,
        status=AnalysisStatus.PENDING,
        input_data={"metadata": req.metadata},
        external_patient_reference=req.patientRef,
        external_hospital_reference=req.hospitalRef,
        external_system=client.client_name,
        requesting_client_id=client.id,
        external_record_kind=req.type,
        external_record_id=req.record_id,
        delivery_status="pending",
    )
    db.add(analysis)
    try:
        db.flush()
    except IntegrityError:
        db.rollback()
        return _find_existing(db, client, req.type, req.record_id), False

    storage = StorageService()
    filename = f"{req.record_id}{fetched.extension}"
    key = storage.generate_storage_key(
        analysis_id=str(analysis.id), filename=filename,
        prefix="images" if req.type == "radiology" else "documents",
    )
    with tempfile.NamedTemporaryFile(delete=False, suffix=fetched.extension) as tmp:
        tmp.write(fetched.content)
        tmp_path = Path(tmp.name)
    try:
        stored = storage.save_file(tmp_path, key)
    finally:
        tmp_path.unlink(missing_ok=True)

    document = UploadedDocument(
        analysis_id=analysis.id,
        filename=key.split("/")[-1],
        original_filename=filename[:255],
        file_type=_DOC_TYPE[fetched.kind_detected],
        file_size=len(fetched.content),
        checksum=hashlib.sha256(fetched.content).hexdigest(),
        storage_key=key,
        storage_provider=stored["storage_provider"],
    )
    db.add(document)
    db.commit()  # must be committed before the worker can see it

def finalize_integration_analysis(db: Session, analysis_id: str, enqueue=None) -> bool:
    analysis = db.query(MedicalAnalysis).filter(MedicalAnalysis.id == UUID(analysis_id)).first()
    if not analysis or analysis.external_record_id is None or analysis.results_expire_at is not None:
        return False

    storage = StorageService()
    for doc in db.query(UploadedDocument).filter_by(analysis_id=analysis.id).all():
        try:
            storage.delete_file(doc.storage_key)
        except Exception as exc:  # a missing file must not block delivery
            logger.warning("source file delete failed", analysis_id=analysis_id, error=type(exc).__name__)
        record_event(db, action="file_deleted", resource_type="analysis", resource_id=analysis_id,
                     outcome="success", client_id=analysis.requesting_client_id)

    analysis.results_expire_at = datetime.utcnow() + timedelta(days=settings.RESULT_RETENTION_DAYS)
    analysis.delivery_status = "pending"
    db.commit()
    (enqueue or _default_enqueue)(analysis_id)
    return True

def backoff_seconds(retries_so_far: int) -> int:
    return settings.CALLBACK_BACKOFF_BASE_SECONDS * (2 ** retries_so_far)


def build_callback_body(db: Session, analysis: MedicalAnalysis) -> bytes:
    failed = analysis.status == AnalysisStatus.FAILED
    id_field = "radiologyId" if analysis.external_record_kind == "radiology" else "documentId"
    payload = {
        "jobId": str(analysis.id),
        "status": "failed" if failed else "completed",
        id_field: analysis.external_record_id,
        "results": None if failed else build_results_payload(db, analysis),
        "error": {"code": "analysis_failed"} if failed else None,
    }
    return json.dumps(payload, sort_keys=True, separators=(",", ":")).encode()


def _http_client() -> httpx.Client:
    return httpx.Client(timeout=settings.CALLBACK_TIMEOUT_SECONDS, follow_redirects=False)


def run_delivery_attempt(db: Session, analysis_id: str, attempt_index: int, http: httpx.Client) -> str:
    analysis = db.query(MedicalAnalysis).filter(MedicalAnalysis.id == UUID(analysis_id)).first()
    if analysis is None:
        return "failed"
    if analysis.delivery_status == "delivered":
        return "delivered"
    client = db.query(IntegrationClient).filter(IntegrationClient.id == analysis.requesting_client_id).first()

    body = build_callback_body(db, analysis)
    timestamp = int(time.time())
    headers = {
        "Content-Type": "application/json",
        "X-ClaraMed-Timestamp": str(timestamp),
        "X-ClaraMed-Signature": sign_payload(client.webhook_secret, timestamp, body),
    }
    error = None
    try:
        response = http.post(client.webhook_url, content=body, headers=headers)
        if not 200 <= response.status_code < 300:
            error = f"http {response.status_code}"
    except httpx.HTTPError as exc:
        error = type(exc).__name__

    analysis.delivery_attempts = (analysis.delivery_attempts or 0) + 1
    if error is None:
        analysis.delivery_status = "delivered"
        analysis.delivery_last_error = None
        analysis.callback_delivered_at = datetime.utcnow()
        outcome = "delivered"
    else:
        analysis.delivery_last_error = error
        outcome = "failed" if attempt_index + 1 >= settings.CALLBACK_MAX_ATTEMPTS else "retry"
        if outcome == "failed":
            analysis.delivery_status = "failed"
    db.commit()
    record_event(db, action="callback_attempt", resource_type="analysis", resource_id=analysis_id,
                 outcome="success" if error is None else "failure", client_id=client.id,
                 client_name=client.client_name, error=error,
                 metadata={"attempt": attempt_index + 1})
    return outcome


@celery_app.task(bind=True, max_retries=settings.CALLBACK_MAX_ATTEMPTS - 1)
def deliver_callback(self, analysis_id: str):
    with get_db_context() as db, _http_client() as http:
        outcome = run_delivery_attempt(db, analysis_id, self.request.retries, http)
    if outcome == "retry":
        raise self.retry(countdown=backoff_seconds(self.request.retries))
    return outcome

    expired = db.query(MedicalAnalysis).filter(
        MedicalAnalysis.results_expire_at.isnot(None),
        MedicalAnalysis.results_expire_at < now,
    ).all()
    if not expired:
        return 0
    storage = StorageService()
    for analysis in expired:
        for doc in db.query(UploadedDocument).filter_by(analysis_id=analysis.id).all():
            try:
                storage.delete_file(doc.storage_key)  # normally already gone; belt and braces
            except Exception:
                pass
        db.delete(analysis)  # cascades to results, documents, findings
    db.commit()
    record_event(db, action="results_purged", resource_type="analysis", resource_id=None,
                 outcome="success", metadata={"count": len(expired)})
    logger.info("purged expired results", count=len(expired))
    return len(expired)

import crypto from "node:crypto";

export function verifyClaraMed(req, rawBody, secret) {
  const ts = req.header("X-ClaraMed-Timestamp");
  const sig = req.header("X-ClaraMed-Signature") || "";
  if (!ts || Math.abs(Date.now() / 1000 - Number(ts)) > 300) return false;
